請用冠廷／老闆的 Google 帳號開 Firebase 規則頁，按「複製全部」貼上後發布。

rules_version = '2';
// 排檯系統 paitai-system — Firestore 安全規則（v12.16）
// 角色：小姐/少爺=1 < 幹部=2 < 經理=3 < 老闆=4 < 特殊（冠廷）=5；tv=唯讀看板（0）；pending=待核准（0）
// 小姐只能走獨立報班頁，不能讀寫營業日主檔；只讀自己的報班／台數，只寫自己的報班紀錄
    // users 文件需帶 rank 欄位且必須等於 rank(role)（供查詢過濾）
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function me() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function hasProfile() { return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)); }
    function active() { return hasProfile() && me().get('disabled', false) != true; }
    function rank(r) { return r == '特殊' ? 5 : r == '老闆' ? 4 : r == '經理' ? 3 : r == '幹部' ? 2 : (r == '少爺' || r == '小姐') ? 1 : 0; }
    function lvl() { return active() ? rank(me().role) : 0; }
    function isTV() { return active() && me().role == 'tv'; }
    function isHostess() { return active() && me().role == '小姐'; }
    function myGirl() { return me().get('girlId', me().get('staffId', '')); }
    function staff() { return lvl() >= 1 && me().role != '小姐' && me().role != 'pending'; }
    function reader() { return staff() || isTV(); }
    function validRole(r) { return r in ['小姐', '少爺', '幹部', '經理', '老闆', '特殊', 'tv', 'kitchen', 'room', 'pending']; }
    function isRoomTab() { return active() && me().role == 'room' && me().get('room', '') != ''; }
    function myRoom() { return me().room; }
    function isKitchen() { return active() && me().role == 'kitchen'; }
    function rankOk(d) { return d.rank == rank(d.role); }

    // ---- v12.14 客人點檯 QR（未登入、持 token 只能動該房）----
    function guestTok(t) {
      return t is string && t.size() >= 16
        && exists(/databases/$(database)/documents/guestTokens/$(t))
        && get(/databases/$(database)/documents/guestTokens/$(t)).data.valid == true
        && request.time.toMillis() < get(/databases/$(database)/documents/guestTokens/$(t)).data.expiresAtMs;
    }
    function guestTokRoom(t) {
      return get(/databases/$(database)/documents/guestTokens/$(t)).data.room;
    }

    // 店家設定、訊息卡、酒單：所有登入人員可讀，經理/老闆可寫
    match /config/settings {
      allow read: if reader() || isHostess();
      allow write: if lvl() >= 3;
    }
    // 首次啟用標記：與第一位老闆帳號同一批寫入，之後不可改
    match /config/boot {
      allow read: if signedIn();
      allow create: if signedIn() && !exists(/databases/$(database)/documents/config/boot)
        && request.resource.data.uid == request.auth.uid
        && getAfter(/databases/$(database)/documents/users/$(request.auth.uid)).data.role in ['特殊', '老闆'];
      allow update, delete: if false;
    }
    // 人員帳號與角色
    match /users/{uid} {
      allow read: if (signedIn() && request.auth.uid == uid) || (lvl() >= 3 && resource.data.rank <= lvl());
      // 第一位帳號（首次啟用）：特殊（冠廷）或老闆
      allow create: if signedIn() && request.auth.uid == uid
        && !exists(/databases/$(database)/documents/config/boot)
        && getAfter(/databases/$(database)/documents/config/boot).data.uid == uid
        && request.resource.data.role in ['特殊', '老闆'] && rankOk(request.resource.data) && request.resource.data.get('disabled', false) == false;
      // Google／Apple 新登入：先建立 pending，等冠廷／老闆／經理核准並綁定身分
      allow create: if signedIn() && request.auth.uid == uid
        && request.resource.data.role == 'pending' && request.resource.data.rank == 0
        && request.resource.data.get('disabled', false) == false
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 40;
      // 老闆以上建立其他人員；不能建立比自己高的角色（只有特殊能建立特殊）
      allow create: if lvl() >= 4 && request.auth.uid != uid && validRole(request.resource.data.role)
        && request.resource.data.role != 'pending'
        && rank(request.resource.data.role) <= lvl() && rankOk(request.resource.data);
      // 本人可改顯示名稱（不能改角色、編號、權限）
      allow update: if signedIn() && request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name'])
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 40;
      // 經理以上改角色或停用；pending 核准時可補員工編號／小姐編號
      allow update: if lvl() >= 3 && request.auth.uid != uid
        && rank(resource.data.role) <= lvl()
        && validRole(request.resource.data.role) && request.resource.data.role != 'pending'
        && rank(request.resource.data.role) <= lvl()
        && (request.resource.data.staffId == resource.data.staffId || resource.data.role == 'pending')
        && rankOk(request.resource.data);
      allow delete: if false;
    }
    // 當日營業資料（每個營業日一份文件）
    match /days/{day} {
      allow read: if reader();
      allow create, update: if staff();
      allow delete: if lvl() >= 3;
    }
    // 歷史歸檔（每日一份）：幹部以上可看；自動日結任何在班人員可建立；修改（手動匯入）幹部以上
    match /hist/{day} {
      allow read: if lvl() >= 2;
      allow create: if staff();
      allow update: if lvl() >= 2;
      allow delete: if lvl() >= 3;
    }
    // 寄酒：幹部以上
    match /wine/{id} {
      allow read, create, update: if lvl() >= 2;
      allow delete: if lvl() >= 3;
    }
    match /wineLog/{id} {
      allow read: if lvl() >= 2;
      allow create: if lvl() >= 2 && request.resource.data.uid == request.auth.uid;
      allow update, delete: if false;
    }
    // 小姐寄放／預支帳本：幹部以上可看、可新增；修改／刪除限經理以上
    match /ledger/{id} {
      allow read: if lvl() >= 2;
      allow create: if lvl() >= 2 && request.resource.data.uid == request.auth.uid;
      allow update, delete: if lvl() >= 3;
    }
    // 員工密碼副本：冠廷（5）可看全部；老闆（4）只能看老闆以下；其他人不能讀。本人改密碼時可寫自己的。
    match /secrets/{uid} {
      function targetRank() { return get(/databases/$(database)/documents/users/$(uid)).data.rank; }
      allow read: if lvl() >= 5 || (lvl() == 4 && targetRank() < 4);
      allow create, update: if (hasProfile() && request.auth.uid == uid)
        || lvl() >= 5 || (lvl() == 4 && targetRank() < 4);
      allow delete: if false;
    }
    // 餐點菜單（v12.2）：登入人員與廚房可讀；經理以上可改
    match /menu/{id} {
      allow read: if reader() || isKitchen() || isRoomTab();
      allow write: if lvl() >= 3;
    }
    // 點餐單（v12.2）：少爺以上點餐；廚房只能改狀態（製作中／已出餐）
    match /orders/{id} {
      allow read: if reader() || isKitchen() || (isRoomTab() && resource.data.room == myRoom()) || (request.auth == null && resource.data.guestToken is string && guestTok(resource.data.guestToken) && resource.data.room == guestTokRoom(resource.data.guestToken));
      // 包廂平板（v12.6）：只能替自己綁定的包廂點餐／點酒
      allow create: if isRoomTab() && request.resource.data.room == myRoom()
        && request.resource.data.createdBy == request.auth.uid && request.resource.data.status == '待做'
        && request.resource.data.fromGuest == true
        && request.resource.data.items is list && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 60
        && request.resource.data.businessDay is string && request.resource.data.createdAt is number;
      allow create: if request.auth == null
        && request.resource.data.guestToken is string
        && guestTok(request.resource.data.guestToken)
        && request.resource.data.room == guestTokRoom(request.resource.data.guestToken)
        && request.resource.data.fromGuest == true
        && request.resource.data.status == '待做'
        && request.resource.data.items is list && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 60
        && request.resource.data.businessDay is string && request.resource.data.createdAt is number;
      allow create: if staff() && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.status == '待做'
        && request.resource.data.room is string && request.resource.data.items is list
        && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 60
        && request.resource.data.businessDay is string && request.resource.data.createdAt is number;
      allow update: if (staff()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'statusAt', 'statusBy', 'items', 'note'])
          && request.resource.data.status in ['待做', '製作中', '已出餐', '取消'])
        // 換包廂（v12.5）：幹部以上可把點單改到新包廂
        || (lvl() >= 2
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['room', 'movedFrom'])
          && request.resource.data.room is string)
        || (isKitchen()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'statusAt', 'statusBy'])
          && request.resource.data.status in ['製作中', '已出餐']
          && request.resource.data.statusBy == request.auth.uid);
      allow delete: if lvl() >= 3;
    }
    // 登入紀錄（v12.3）：本人建立自己的紀錄；冠廷（5）看全部、可刪；老闆（4）只看老闆以下
    match /loginLogs/{id} {
      allow create: if hasProfile() && request.resource.data.uid == request.auth.uid
        && request.resource.data.rank == rank(me().role)
        && request.resource.data.atMs is number;
      allow read: if lvl() >= 5 || (lvl() == 4 && resource.data.rank < 4);
      allow update: if false;
      allow delete: if lvl() >= 5;
    }
    // 包廂平板（v12.6）：可點小姐清單（只有編號、名字、照片）
    match /public/avail {
      allow read: if reader() || isRoomTab();
      allow write: if staff();
    }
    // 包廂平板（v12.6）：客人需求（點小姐／呼叫 少爺/公主）
    match /guestReq/{id} {
      allow read: if reader() || (isRoomTab() && resource.data.room == myRoom()) || (request.auth == null && resource.data.guestToken is string && guestTok(resource.data.guestToken) && resource.data.room == guestTokRoom(resource.data.guestToken));
      allow create: if isRoomTab() && request.resource.data.room == myRoom()
        && request.resource.data.createdBy == request.auth.uid && request.resource.data.status == '已送出'
        && request.resource.data.type in ['點小姐', '呼叫', '續番選擇']
        && request.resource.data.girls is list && request.resource.data.girls.size() <= 10
        && request.resource.data.businessDay is string && request.resource.data.createdAt is number;
      // 客人掃 QR（未登入）：只能替 token 綁定的包廂送出需求
      allow create: if request.auth == null
        && request.resource.data.guestToken is string
        && guestTok(request.resource.data.guestToken)
        && request.resource.data.room == guestTokRoom(request.resource.data.guestToken)
        && request.resource.data.fromGuest == true
        && request.resource.data.status == '已送出'
        && request.resource.data.type in ['點小姐', '呼叫', '續番選擇']
        && request.resource.data.girls is list && request.resource.data.girls.size() <= 10
        && request.resource.data.businessDay is string && request.resource.data.createdAt is number;
      allow update: if staff()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'handledBy', 'handledByName', 'handledAt'])
        && request.resource.data.status in ['已確認', '已拒絕', '已處理'];
      allow delete: if lvl() >= 3;
    }
    // 操作紀錄：只能新增，不能改不能刪


    match /guestTokens/{t} {
      // 文件 ID 即密鑰：允許單筆讀取（不可 list）；過期或作廢仍可讀以顯示失效原因
      allow get: if true;
      allow list: if staff() || isRoomTab();
      allow create: if (lvl() >= 2 || isRoomTab())
        && request.resource.data.token == t
        && request.resource.data.room is string
        && request.resource.data.valid == true
        && request.resource.data.expiresAtMs is number
        && (!isRoomTab() || request.resource.data.room == myRoom());
      allow update: if lvl() >= 2
        || (isRoomTab() && resource.data.room == myRoom() && request.resource.data.room == myRoom());
      allow delete: if lvl() >= 3;
      match /view/{id} {
        allow read: if guestTok(t) || staff() || isRoomTab();
        allow write: if staff() || (isRoomTab() && get(/databases/$(database)/documents/guestTokens/$(t)).data.room == myRoom());
      }
    }
    // ---- v12.14 小姐自助報班（獨立頁；自動匯入今日排檯）----
    match /girlCheckin/{id} {
      allow read: if staff() || isRoomTab()
        || (isHostess() && resource.data.girlId == myGirl());
      allow create: if isHostess()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.girlId == myGirl()
        && request.resource.data.day is string
        && request.resource.data.on is bool;
      allow create: if lvl() >= 2
        && request.resource.data.girlId is string
        && request.resource.data.day is string
        && request.resource.data.on is bool
        && request.resource.data.src in ['staff', 'test'];
      allow update: if isHostess()
        && resource.data.girlId == myGirl() && resource.data.uid == request.auth.uid
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.girlId == myGirl()
        && request.resource.data.day == resource.data.day
        && request.resource.data.on is bool;
      allow update: if lvl() >= 2
        && request.resource.data.girlId is string
        && request.resource.data.on is bool;
      // 幹部處理「已有番不能自行取消」等標記
      allow update: if lvl() >= 2;
      allow delete: if isHostess() && resource.data.girlId == myGirl() && resource.data.uid == request.auth.uid
        || lvl() >= 3;
    }
    // 小姐只看自己的台數／歷史（由排檯端寫入，不經手營業日主檔）
    match /girlHist/{id} {
      allow read: if staff() || (isHostess() && resource.data.girlId == myGirl());
      allow create, update: if staff();
      allow delete: if lvl() >= 3;
    }
    // 少爺／公主打卡（含 GPS）；本人讀自己的，幹部以上讀全部並可標記審核
    match /punches/{id} {
      allow read: if staff() && (lvl() >= 2 || resource.data.uid == request.auth.uid);
      allow create: if staff() && request.resource.data.uid == request.auth.uid
        && request.resource.data.type in ['in', 'out']
        && request.resource.data.day is string
        && request.resource.data.lat is number
        && request.resource.data.lng is number;
      allow update: if lvl() >= 2
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['review', 'reviewBy', 'reviewByName', 'reviewNote', 'reviewAt']);
      allow delete: if false;
    }
    // 請假：本人申請／撤回；幹部以上核准或駁回
    match /leaves/{id} {
      allow read: if staff() && (lvl() >= 2 || resource.data.uid == request.auth.uid);
      allow create: if staff() && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == '待核准'
        && request.resource.data.start is string && request.resource.data.end is string;
      allow update: if staff() && resource.data.uid == request.auth.uid
        && resource.data.status == '待核准'
        && request.resource.data.status == '已取消'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'updatedAt']);
      allow update: if lvl() >= 2
        && resource.data.status == '待核准'
        && request.resource.data.status in ['已核准', '已駁回']
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'decidedBy', 'decidedByName', 'decisionNote', 'decidedAt', 'updatedAt']);
      allow delete: if false;
    }
    match /audit/{id} {
      allow read: if lvl() >= 3;
      allow create: if staff() && request.resource.data.uid == request.auth.uid;
      allow update, delete: if false;
    }
  }
}
