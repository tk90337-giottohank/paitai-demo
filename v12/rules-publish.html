rules_version = '2';
// 排檯系統 paitai-system — Firestore 安全規則（v12）
// 角色：小姐/少爺=1 < 幹部=2 < 經理=3 < 老闆=4 < 特殊（冠廷）=5；tv=唯讀看板（0）
    // users 文件需帶 rank 欄位且必須等於 rank(role)（供查詢過濾）
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function me() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function hasProfile() { return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)); }
    function active() { return hasProfile() && me().get('disabled', false) != true; }
    function rank(r) { return r == '特殊' ? 5 : r == '老闆' ? 4 : r == '經理' ? 3 : r == '幹部' ? 2 : (r == '少爺' || r == '小姐') ? 1 : 0; }
    function lvl() { return active() ? rank(me().role) : 0; }
    function isTV() { return active() && me().role == 'tv'; }
    function staff() { return lvl() >= 1; }
    function reader() { return staff() || isTV(); }
    function validRole(r) { return r in ['小姐', '少爺', '幹部', '經理', '老闆', '特殊', 'tv', 'kitchen', 'room']; }
    function isRoomTab() { return active() && me().role == 'room' && me().get('room', '') != ''; }
    function myRoom() { return me().room; }
    function isKitchen() { return active() && me().role == 'kitchen'; }
    function rankOk(d) { return d.rank == rank(d.role); }

    // 店家設定、訊息卡、酒單：所有登入人員可讀，經理/老闆可寫
    match /config/settings {
      allow read: if reader();
      allow write: if lvl() >= 3;
    }
    // 首次啟用標記：與第一位老闆帳號同一批寫入，之後不可改
    match /config/boot {
      allow read: if signedIn();
      allow create: if signedIn() && !exists(/databases/$(database)/documents/config/boot)
        && request.resource.data.uid == request.auth.uid
        && getAfter(/databases/$(database)/documents/users/$(request.auth.uid)).data.role in ['特殊', '老闆'];
      allow update, delete: if false;
    }
    // 人員帳號與角色
    match /users/{uid} {
      allow read: if (signedIn() && request.auth.uid == uid) || (lvl() >= 3 && resource.data.rank <= lvl());
      // 第一位帳號（首次啟用）：特殊（冠廷）或老闆
      allow create: if signedIn() && request.auth.uid == uid
        && !exists(/databases/$(database)/documents/config/boot)
        && getAfter(/databases/$(database)/documents/config/boot).data.uid == uid
        && request.resource.data.role in ['特殊', '老闆'] && rankOk(request.resource.data) && request.resource.data.get('disabled', false) == false;
      // 老闆以上建立其他人員；不能建立比自己高的角色（只有特殊能建立特殊）
      allow create: if lvl() >= 4 && request.auth.uid != uid && validRole(request.resource.data.role)
        && rank(request.resource.data.role) <= lvl() && rankOk(request.resource.data);
      // 經理/老闆改角色或停用：不能改自己、不能改比自己高的人、不能給超過自己的角色
      allow update: if lvl() >= 3 && request.auth.uid != uid
        && rank(resource.data.role) <= lvl()
        && validRole(request.resource.data.role) && rank(request.resource.data.role) <= lvl()
        && request.resource.data.staffId == resource.data.staffId && rankOk(request.resource.data);
      allow delete: if false;
    }
    // 當日營業資料（每個營業日一份文件）
    match /days/{day} {
      allow read: if reader();
      allow create, update: if staff();
      allow delete: if lvl() >= 3;
    }
    // 歷史歸檔（每日一份）：幹部以上可看；自動日結任何在班人員可建立；修改（手動匯入）幹部以上
    match /hist/{day} {
      allow read: if lvl() >= 2;
      allow create: if staff();
      allow update: if lvl() >= 2;
      allow delete: if lvl() >= 3;
    }
    // 寄酒：幹部以上
    match /wine/{id} {
      allow read, create, update: if lvl() >= 2;
      allow delete: if lvl() >= 3;
    }
    match /wineLog/{id} {
      allow read: if lvl() >= 2;
      allow create: if lvl() >= 2 && request.resource.data.uid == request.auth.uid;
      allow update, delete: if false;
    }
    // 小姐寄放／預支帳本：幹部以上可看、可新增；修改／刪除限經理以上
    match /ledger/{id} {
      allow read: if lvl() >= 2;
      allow create: if lvl() >= 2 && request.resource.data.uid == request.auth.uid;
      allow update, delete: if lvl() >= 3;
    }
    // 員工密碼副本：冠廷（5）可看全部；老闆（4）只能看老闆以下；其他人不能讀。本人改密碼時可寫自己的。
    match /secrets/{uid} {
      function targetRank() { return get(/databases/$(database)/documents/users/$(uid)).data.rank; }
      allow read: if lvl() >= 5 || (lvl() == 4 && targetRank() < 4);
      allow create, update: if (hasProfile() && request.auth.uid == uid)
        || lvl() >= 5 || (lvl() == 4 && targetRank() < 4);
      allow delete: if false;
    }
    // 餐點菜單（v12.2）：登入人員與廚房可讀；經理以上可改
    match /menu/{id} {
      allow read: if reader() || isKitchen() || isRoomTab();
      allow write: if lvl() >= 3;
    }
    // 點餐單（v12.2）：少爺以上點餐；廚房只能改狀態（製作中／已出餐）
    match /orders/{id} {
      allow read: if reader() || isKitchen() || (isRoomTab() && resource.data.room == myRoom());
      // 包廂平板（v12.6）：只能替自己綁定的包廂點餐／點酒
      allow create: if isRoomTab() && request.resource.data.room == myRoom()
        && request.resource.data.createdBy == request.auth.uid && request.resource.data.status == '待做'
        && request.resource.data.fromGuest == true
        && request.resource.data.items is list && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 60
        && request.resource.data.businessDay is string && request.resource.data.createdAt is number;
      allow create: if staff() && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.status == '待做'
        && request.resource.data.room is string && request.resource.data.items is list
        && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 60
        && request.resource.data.businessDay is string && request.resource.data.createdAt is number;
      allow update: if (staff()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'statusAt', 'statusBy', 'items', 'note'])
          && request.resource.data.status in ['待做', '製作中', '已出餐', '取消'])
        // 換包廂（v12.5）：幹部以上可把點單改到新包廂
        || (lvl() >= 2
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['room', 'movedFrom'])
          && request.resource.data.room is string)
        || (isKitchen()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'statusAt', 'statusBy'])
          && request.resource.data.status in ['製作中', '已出餐']
          && request.resource.data.statusBy == request.auth.uid);
      allow delete: if lvl() >= 3;
    }
    // 登入紀錄（v12.3）：本人建立自己的紀錄；冠廷（5）看全部、可刪；老闆（4）只看老闆以下
    match /loginLogs/{id} {
      allow create: if hasProfile() && request.resource.data.uid == request.auth.uid
        && request.resource.data.rank == rank(me().role)
        && request.resource.data.atMs is number;
      allow read: if lvl() >= 5 || (lvl() == 4 && resource.data.rank < 4);
      allow update: if false;
      allow delete: if lvl() >= 5;
    }
    // 包廂平板（v12.6）：可點小姐清單（只有編號、名字、照片）
    match /public/avail {
      allow read: if reader() || isRoomTab();
      allow write: if staff();
    }
    // 包廂平板（v12.6）：客人需求（點小姐／呼叫 少爺/公主）
    match /guestReq/{id} {
      allow read: if reader() || (isRoomTab() && resource.data.room == myRoom());
      allow create: if isRoomTab() && request.resource.data.room == myRoom()
        && request.resource.data.createdBy == request.auth.uid && request.resource.data.status == '已送出'
        && request.resource.data.type in ['點小姐', '呼叫']
        && request.resource.data.girls is list && request.resource.data.girls.size() <= 10
        && request.resource.data.businessDay is string && request.resource.data.createdAt is number;
      allow update: if staff()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'handledBy', 'handledByName', 'handledAt'])
        && request.resource.data.status in ['已確認', '已拒絕', '已處理'];
      allow delete: if lvl() >= 3;
    }
    // 操作紀錄：只能新增，不能改不能刪
    match /audit/{id} {
      allow read: if lvl() >= 3;
      allow create: if staff() && request.resource.data.uid == request.auth.uid;
      allow update, delete: if false;
    }
  }
}
